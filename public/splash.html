use tauri::{WebviewUrl, WebviewWindowBuilder};


#[cfg_attr(mobile, tauri::mobile_entry_point)]

pub fn run() {

    tauri::Builder::default()

        .setup(|app| {


            // SPLASH

            let splash = WebviewWindowBuilder::new(

                app,

                "splash",

                WebviewUrl::App("splash.html".into()),

            )

            .title("Loading...")

            .build()?;


            // MAIN (Next.js)

            let main = WebviewWindowBuilder::new(

                app,

                "main",

                WebviewUrl::External("http://localhost:9002".parse().unwrap()),

            )

            .title("RiderMaster Pro")

            .build()?;


            // cerrar splash cuando main envíe "tauri://ready"

            let splash_ref = splash.clone();

            main.once("tauri://ready", move |_| {

                let _ = splash_ref.close();

            });


            Ok(())

        })

        .run(tauri::generate_context!())

        .expect("error while running app");

}
